% The generic package must be usable without a manual margin recipe.
\documentclass{beamer}
\usetheme{Madrid}
\usepackage{beamer-floral}
\makeatletter
\ifdim\beamer@leftmargin=2.5cm\else\errmessage{Missing floral left margin default}\fi
\ifdim\beamer@rightmargin=1cm\else\errmessage{Missing floral right margin default}\fi
\makeatother
% Explicit document choices after loading must continue to work.
\setbeamersize{text margin left=3cm,text margin right=1.2cm}
\makeatletter
\ifdim\beamer@leftmargin=3cm\else\errmessage{Custom left margin was ignored}\fi
\ifdim\beamer@rightmargin=1.2cm\else\errmessage{Custom right margin was ignored}\fi
\makeatother
\florasetup{seed=3,density=0.3}
\begin{document}
\begin{frame}{Room to grow}
  Generous defaults leave space for the border. Document settings can give it more.
\end{frame}
\end{document}
