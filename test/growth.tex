\documentclass[aspectratio=169]{beamer}
\usetheme{floral}
\florasetup{seed=3,density=0.2,growth=overlays}
\newcount\executions
\directlua{
  local adapter=beamer_floral.render_page
  local seen={}
  beamer_floral.render_page=function(spec,a)
    local pt=adapter(spec,a)
    if os.getenv('FLORAL_VERIFY_STABLE')=='1' then
      local id=spec.page
      local k,n=beamer_floral.stage,beamer_floral.stages
      local expected=({1,2,4,16,64,2,2,2,2,1,1,2,1})[id]
      if n~=expected then tex.error('Incorrect displayed overlay count for frame '..id..': '..n) end
      if pt then
        local body=pt:body()
        if seen[id]==body then tex.error('Repeated growth stage in frame '..id) end
        seen[id]=body
      end
    end
    return pt
  end
}
