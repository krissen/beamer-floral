\documentclass[aspectratio=169]{beamer}
\usetheme{floral}
\florasetup{seed=3,density=0.25}
\setbeamertemplate{section page}[floral]
\AtBeginSection[]{\begin{frame}[plain]\sectionpage\end{frame}}
\title{Section pages}
\author{The Floral Project}
\date{}
\directlua{
  local prior=beamer_floral.prepare
  local function document(spec,why)
    assert(spec.layout=='surround' and spec.mix.density==0.25,why)
  end
  local function section(spec,why)
    assert(spec.layout=='left' and spec.mix.density==1 and spec.mix.sprays==2,why)
  end
  beamer_floral.prepare=function(spec)
    prior(spec)
    local n=beamer_floral.count
    local class=beamer_floral.frame.class
    if n==1 then
      assert(spec.mix.density==2,'title page keeps the title preset')
    elseif n==2 then
      section(spec,'an AtBeginSection page takes the section preset')
      assert(class=='section','an AtBeginSection page is a section page')
    elseif n==3 then
      document(spec,'section defaults must not leak')
    elseif n==4 then
      assert(spec.layout=='right' and spec.mix.density==0.5,'frame override must win')
    elseif n==5 then
      document(spec,'another section page template keeps the document settings')
      assert(class=='section','another template is still a section page')
    elseif n==6 then
      section(spec,'first-overlay section preset must persist')
    elseif n==7 then
      section(spec,'a titled frame can carry a section page')
    end
  end
}
\begin{document}

\section{A first section}

\florathisframe{layout=right,density=0.5}

  \sectionpage
\end{frame}

  \only<1>{\sectionpage}
  \only<2>{Content following the section page keeps the same garden.}
\end{frame}
\begin{frame}\sectionpage\end{frame}
\end{document}
