\documentclass[aspectratio=169]{beamer}
\usetheme{floral}
\florasetup{seed=3,density=0.3}
\directlua{
 local original=flora.page
 flora.page=function(a)
 local effective=require('flora.settings').merge(flora.settings,flora.once)
 texio.write_nl('term and log','FLORAL-STATE page='..a.page..' layout='..tostring(effective.layout)..' density='..tostring(effective.mix.density)..' x0='..a.text_x0..' title='..a.frametitle_height)
 return original(a)
 end
}
